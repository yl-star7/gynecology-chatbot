# 모아애착 챗봇 플로우 Mermaid 정리

## 1. 공통 설계 구조

```mermaid
flowchart TD
    A["사용자 접속"] --> B["간호사 캐릭터 등장<br/>상황에 맞는 표정 이미지 표시"]
    B --> C["감정 확인<br/>선택지 + 직접 입력"]
    C --> D{"사용자가 지금 확인할까요?"}
    D -- "아니요, 이따가 확인할래요" --> Z["대화 종료<br/>이따가 다시 만나요"]
    D -- "네" --> E["태아 발달 정보<br/>주수별 아기 이미지 + 크기 + 발달 + 아기의 말"]
    E --> F{"엄마의 변화도 확인할까요?"}
    F -- "아니요, 이따가 확인할래요" --> Z
    F -- "네" --> G["모체 변화 정보<br/>주수별 몸 변화와 증상 안내"]
    G --> H{"오늘 실천할 일을 볼까요?"}
    H -- "아니요, 이따가 확인할래요" --> Z
    H -- "네" --> I["생활 체크리스트<br/>3개 실천 항목 표시"]
    I --> J["체크 완료<br/>웹 재접속 후에도 체크 가능"]
    J --> K["모아애착 질문 선택<br/>2개 질문 중 선택 또는 둘 다 답변"]
    K --> L["AI 기반 공감 대화<br/>감정 반영 + 격려 + 애착 의미화"]
    L --> M["마무리 인사<br/>내일 다시 만나요"]

    N["AI / 생성형 사용 영역"] -.-> C
    N -.-> L
    O["주수별 이미지 풀"] -.-> E
    P["대화 내용 달력 저장"] -.-> M
```

## 2. 페르소나별 전체 흐름

```mermaid
flowchart TD
    Start["모아애착 챗봇 시작"] --> Persona{"사용자 유형 / 임신 주수"}

    Persona --> P1["불안형 예비맘<br/>20주 3일차"]
    Persona --> P2["긍정·적극형 예비맘<br/>24주 1일차"]
    Persona --> P3["내성적·고독형 예비맘<br/>32주 4일차"]
    Persona --> P4["실용·정보추구형 예비맘<br/>28주 2일차"]

    P1 --> P1Emotion["감정 확인<br/>좋아요 / 우울해요 / 슬퍼요 / 화나요 / 직접 입력"]
    P1Emotion --> P1Branch{"대표 분기"}
    P1Branch -- "직접 입력: 날씨가 좋아서 기분도 좋아요" --> P1Positive["밝은 표정<br/>아기 성장 확인 제안"]
    P1Branch -- "우울해요" --> P1Low["부드러운 표정<br/>마음을 무겁게 한 이유 확인"]
    P1Low --> P1LowReason{"우울 이유"}
    P1LowReason -- "그냥 이유 없이 그래요" --> P1Empathy1["이유 없이 그런 날도 있다고 공감"]
    P1LowReason -- "몸이 너무 피곤해요" --> P1Empathy2["피곤함을 인정하고 아기 성장 확인으로 연결"]

    P2 --> P2Emotion["감정 확인<br/>좋아요 / 신나요 / 피곤해요 / 슬퍼요 / 직접 입력"]
    P2Emotion --> P2Branch{"대표 분기"}
    P2Branch -- "피곤해요" --> P2Tired["걱정스러운 표정<br/>가볍게 아기 소식만 확인 제안"]
    P2Branch -- "신나요" --> P2Happy["밝은 표정<br/>좋은 일 확인"]
    P2Happy --> P2Reason{"신난 이유"}
    P2Reason -- "아기 발차기가 느껴졌어요" --> P2Kick["태동을 특별한 순간으로 의미화"]

    P3 --> P3Emotion["감정 확인<br/>좋아요 / 그냥 그래요 / 피곤해요 / 우울해요 / 직접 입력"]
    P3Emotion --> P3Branch{"대표 분기"}
    P3Branch -- "그냥 그래요" --> P3Neutral["차분한 표정<br/>가볍게 아기 소식 확인 제안"]
    P3Branch -- "우울해요" --> P3Low["걱정스럽고 따뜻한 표정<br/>마음을 무겁게 한 이유 확인"]
    P3Low --> P3Reason{"우울 이유"}
    P3Reason -- "혼자인 것 같아요" --> P3Lonely["외로움을 이해하고 아기의 말 듣기로 연결"]

    P4 --> P4Emotion["감정 확인<br/>좋아요 / 궁금한 게 있어요 / 피곤해요 / 걱정돼요 / 직접 입력"]
    P4Emotion --> P4Branch{"대표 분기"}
    P4Branch -- "궁금한 게 있어요" --> P4Question["집중하는 표정<br/>궁금한 주제 확인"]
    P4Question --> P4Topic{"정보 주제"}
    P4Topic -- "태동 횟수가 정상인지" --> P4KickCount["28주 이후 태동 기준 안내<br/>발달 정보 확인 제안"]
    P4Branch -- "걱정돼요" --> P4Worry["공감하는 표정<br/>걱정 주제 확인"]
    P4Worry --> P4WorryReason{"걱정 이유"}
    P4WorryReason -- "아기가 잘 자라고 있는지" --> P4Data["데이터 기반 확인으로 안심 유도"]

    P1Positive --> Common["공통 정보 플로우"]
    P1Empathy1 --> Common
    P1Empathy2 --> Common
    P2Tired --> Common
    P2Kick --> Common
    P3Neutral --> Common
    P3Lonely --> Common
    P4KickCount --> Common
    P4Data --> Common

    Common --> Baby["태아 발달 정보"]
    Baby --> Mother["모체 변화 정보"]
    Mother --> Checklist["생활 체크리스트"]
    Checklist --> Reflection["모아애착 질문"]
    Reflection --> Gpt["AI 공감 대화"]
    Gpt --> End["일일 대화 종료"]
```
